main :              NEWLINE* (toplvl-statement? NEWLINE+)*

toplvl-statement :  class-expr
                    statement

statements :        NEWLINE* (statement? NEWLINE+)*

statement :         KEYWORD:"return" (expr (SEPARATOR expr)*)?
                    KEYWORD:"break"
                    KEYWORD:"continue"
                    alloc-expr
                    if-expr
                    nfor-expr
                    gfor-expr
                    while-expr
                    repeat-expr
                    expr

expr :              type-specifiers IDENTIFIER EQ expr
                    logical (EQ expr)?
                    import-expr

logical :           (comparaison ((KEYWORD:"and" | KEYWORD:"or" | "||" | "&&" | "|" | "&") comparaison)*)
                
comparaison :       KEYWORD:'not' comparaison
                    arithmetics ((EEQ || NEQ || LT || GT || LTEQ || GTEQ) arithmetics)*
        
arithmetics :       term ((PLUS|MINUS|CONCAT) term)*

term :              factor ((MUL|DIV|IDIV|MOD) factor)*

factor :            (PLUS|MINUS) factor
                    power

power :             call ((POW) factor)*

postfix :           numerics ( postfix_op )*

postfix_op :        COLON? LPAR (expr (SEPARATOR expr)*)? RPAR
                    LSBRAC logical RSBRAC
                    DOT postfix

numerics :          FLOAT
                    STR
                    IDENTIFIER
                    LPAR expr RPAR
                    table-expr
                    func-expr


table-expr :        LCBRAC (expr (SEPARATOR expr)*)? RCBRAC

alloc-expr :        KEYWORD:"alloc" expr ARROW (expr (SEPARATOR expr)*)? KEYWORD:"end"


if-expr :           KEYWORD:"if" expr KEYWORD:"then" statements
                    (KEYWORD:"elseif" expr KEYWORD:"then" statements)*
                    (KEYWORD:"else" statements)*
                    KEYWORD:"end"


nfor-expr :         KEYWORD:"for" IDENTIFIER EQ expr SEPARATOR expr
                    (SEPARATOR expr)?
                    KEYWORD:"repeat"
                    statements
                    KEYWORD:"end"

gfor-expr :         KEYWORD:"for" IDENTIFIER (SEPARATOR IDENTIFIER)?
                    KEYWORD:"in" expr
                    KEYWORD:"repeat"
                    statements
                    KEYWORD:"end"


while-expr :        KEYWORD:"while" expr KEYWORD:"repeat"
                    statements
                    KEYWORD:"end"

repeat-expr :       KEYWORD:"repeat"
                    statements
                    KEYWORD:"while" expr

import-expr :       KEYWORD:"import" STR (KEYWORD:"as" IDENTIFIER)? KEYWORD:"end"

func-expr :         KEYWORD:"func" (type-specifiers? IDENTIFIER)?
                    LPAR (type-specifiers? IDENTIFIER (SEPARATOR type-specifiers? IDENTIFIER)*)? RPAR
                    statements
                    KEYWORD:"end"

class-expr :        KEYWORD:"class" IDENTIFIER
                    (ARROW IDENTIFIER (SEPARATOR IDENTIFIER)*)?
                    main KEYWORD:"end"

type-specifiers :   KEYWORD:"var"
                    KEYWORD:"num"
                    KEYWORD:"str"
                    KEYWORD:"table"
                    IDENTIFIER
