# math.txt — standard math library functions

# Basic arithmetic
math.abs(x)
math.floor(x)
math.ceil(x)
math.round(x)
math.trunc(x)
math.fmod(x, y)

# Comparison
math.max(x, y)
math.min(x, y)
math.clamp(x, lo, hi)

# Powers
math.pow(x, y)
math.sqrt(x)

# Trigonometry
math.sin(x)
math.cos(x)
math.tan(x)
math.asin(x)
math.acos(x)
math.atan(x)
math.atan2(y, x)

# Exponential & logarithms
math.exp(x)
math.log(x)        # natural log
math.log10(x)

# Random
math.random()      # 0..1
math.random_int(lo, hi)

# Constants
math.pi
math.e

# Utility
math.sign(x)
math.is_nan(x)
math.is_inf(x)
math.deg(x)        # radians → degrees
math.rad(x)        # degrees → radians
